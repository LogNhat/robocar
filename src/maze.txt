// Định nghĩa chân XSHUT điều khiển khởi động độc lập cho 3 cảm biến VL53L0X
#define XSHUT_CENTER 15
#define XSHUT_LEFT   17
#define XSHUT_RIGHT  16

// Địa chỉ I2C tùy biến sau khi gán địa chỉ mới cho 3 cảm biến VL53L0X

// ============================================================================
// CẤU HÌNH SAI SỐ HIỆU CHUẨN CẢM BIẾN LASER (CALIBRATION OFFSETS IN MM)

// Đơn vị: mm. Giá trị này được cộng vào kết quả đo thô (để bù trừ nếu cảm biến đo lệch).
// Ví dụ: Nếu cảm biến đo 105mm nhưng thực tế thước đo là 100mm, đặt offset là -5.
const int OFFSET_CENTER = 0;  // Sai số cảm biến GIỮA
const int OFFSET_LEFT   = 0;  // Sai số cảm biến TRÁI (Lắp xiên góc 45°)
const int OFFSET_RIGHT  = 0;  // Sai số cảm biến PHẢI (Lắp xiên góc 45°)


// Định nghĩa các chân Driver Động cơ TB6612FNG (Lấy từ th.text)
const int PWMA = 27; // Điều khiển tốc độ bánh Phải (Kênh A)
const int AIN1 = 18; // Chiều bánh Phải
const int AIN2 = 19; // Chiều bánh Phải

const int PWMB = 13; // Điều khiển tốc độ bánh Trái (Kênh B)
const int BIN1 = 5;  // Chiều bánh Trái
const int BIN2 = 23; // Chiều bánh Trái

Adafruit_VL53L0X lox_right  = Adafruit_VL53L0X();

// ----------------------------------------------------------------------------
// THÔNG SỐ VẬT LÝ XE VÀ MÊ CUNG (KÍCH THƯỚC Ô MAZE: 20cm x 20cm = 200mm x 200mm)
// ----------------------------------------------------------------------------
const float CAR_WIDTH  = 120.0; // Chiều rộng xe: 12cm = 120mm
const float CAR_LENGTH = 140.0; // Chiều dài xe: 14cm = 140mm

// ----------------------------------------------------------------------------
// TOÁN HỌC CỦA CẢM BIẾN LẮP GÓC ĐỘ 45 ĐỘ
// ----------------------------------------------------------------------------
// Cảm biến Trái & Phải lắp góc 45 độ hướng ra ngoài so với phương thẳng đứng của xe.
// - Khi xe đi song song với tường:
//   Góc xiên phi = 45 độ. Khoảng cách vuông góc tới tường (d_perp) được tính bằng:
//   d_perp = Khoảng_Cách_Đo_Được * sin(45°) = Khoảng_Cách_Đo_Được * 0.707
// - Nếu xe đi lệch hoặc ở giữa ô:
//   Khi xe ở chính giữa ô (độ rộng đường đi 200mm):
//   Khoảng cách vuông góc từ thành xe tới tường là d_perp = (200mm - 120mm) / 2 = 40mm.
//   Khoảng cách cảm biến 45 độ đo được lý tưởng là: 40mm / sin(45°) = 40 * 1.414 = 56.6mm.
//   If xe lệch chạm sát tường đối diện, d_perp tối đa là 200mm - 120mm = 80mm.
//   Khoảng cách cảm biến đo được tối đa khi CÓ tường bên cạnh là: 80mm / sin(45°) = 80 * 1.414 = 113.1mm.
// - Do đó, nếu đọc được khoảng cách đo bên hông < 150mm => Chắc chắn CÓ TƯỜNG bên hông.
//   Nếu đọc được khoảng cách đo bên hông >= 150mm => Trực quan là hướng đó ĐANG MỞ (Không có tường).
// ----------------------------------------------------------------------------
const int WALL_THRESHOLD_SIDE   = 150; // Ngưỡng phát hiện tường hông (45 độ)
const int WALL_THRESHOLD_CENTER = 150; // Ngưỡng phát hiện tường trước (0 độ)

// Các thông số điều khiển động cơ (Sẽ được giả lập log)
const int BASE_SPEED = 150;   
const int TURN_SPEED = 140;   
const unsigned long MOVE_FORWARD_DURATION = 800; // ms

// Hệ số PD của thuật toán giữ thẳng bằng Gyro
float gyroKp = 3.5;
float gyroKd = 0.6;

// Biến lưu trạng thái góc và điều hướng
float gyroZOffset = 0.0;
float currentHeading = 0.0;
unsigned long lastGyroTime = 0;

// Các trạng thái của xe trong mê cung

  STATE_DECIDE,
  STATE_MOVE_FORWARD,
  STATE_TURN_LEFT,
  STATE_TURN_RIGHT,
  STATE_TURN_AROUND
};

RobotState currentState = STATE_DECIDE;

void setupMotors() {
  // Chỉ cấu hình chân GPIO làm đầu ra, không kích hoạt công suất
  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);
  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);

#if ESP_ARDUINO_VERSION_MAJOR >= 3
  pinMode(PWMA, OUTPUT);
  pinMode(PWMB, OUTPUT);
#else
  ledcSetup(0, 5000, 8);
  ledcAttachPin(PWMA, 0);
  ledcSetup(1, 5000, 8);
  ledcAttachPin(PWMB, 1);
#endif
}

  speed = constrain(speed, -255, 255);
  /* Đã comment để giả lập
  if (speed >= 0) {
    digitalWrite(BIN1, HIGH);
    digitalWrite(BIN2, LOW);
  } else {
    digitalWrite(BIN1, LOW);
    digitalWrite(BIN2, HIGH);
    speed = -speed;
  }
  #if ESP_ARDUINO_VERSION_MAJOR >= 3
    analogWrite(PWMB, speed);
  #else
    ledcWrite(1, speed);
  #endif
  */
}

void setMotorRight(int speed) {
  speed = constrain(speed, -255, 255);
  /* Đã comment để giả lập
  if (speed >= 0) {
    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, LOW);
  } else {
    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, HIGH);
    speed = -speed;
  }
  #if ESP_ARDUINO_VERSION_MAJOR >= 3
    analogWrite(PWMA, speed);
  #else
    ledcWrite(0, speed);
  #endif
  */
}

void stopMotors() {
  setMotorLeft(0);
  setMotorRight(0);
}

bool isDevicePresent(byte address) {
  Wire.beginTransmission(address);
  return (Wire.endTransmission() == 0);
}

void initLaserSensors() {
  Serial.println("\n[SYSTEM] Bắt đầu khởi động các cảm biến Laser (Chế độ Chủ động OUTPUT)...");

  // Bước 1: Kéo tất cả chân XSHUT xuống LOW làm OUTPUT để đưa toàn bộ cảm biến vào Standby
  pinMode(XSHUT_CENTER, OUTPUT);
  pinMode(XSHUT_LEFT, OUTPUT);
  pinMode(XSHUT_RIGHT, OUTPUT);
  
  digitalWrite(XSHUT_CENTER, LOW);
  digitalWrite(XSHUT_LEFT, LOW);
  digitalWrite(XSHUT_RIGHT, LOW);
  delay(150); // Chờ 150ms để xả hết tụ điện và tắt hẳn các cảm biến

  // --------------------------------------------------------------------------
  // BƯỚC 2: Khởi động cảm biến CENTER (Ở giữa)
  // Chủ động kéo HIGH làm OUTPUT để tránh chân bị trôi lơ lửng gây khóa bus I2C
  // --------------------------------------------------------------------------
  pinMode(XSHUT_CENTER, OUTPUT);
  digitalWrite(XSHUT_CENTER, HIGH);
  delay(450); // Chờ 450ms cho cảm biến Center khởi động ổn định tuyệt đối

  if (!isDevicePresent(0x29)) {
    Serial.println("❌ LỖI VẬT LÝ: Cảm biến CENTER không thức dậy ở địa chỉ 0x29!");
    Serial.println("👉 Vui lòng kiểm tra lại tiếp xúc chân XSHUT_CENTER (GPIO 17) hoặc nguồn VCC/GND!");
    while (1) delay(200);
  }

  if (!lox_center.begin(ADDR_CENTER, false, &Wire)) {
    Serial.println("❌ Lỗi: Không thể cấu hình Center ở địa chỉ 0x30");
    while (1) delay(100);
  }
  Serial.println("✓ Khởi động thành công cảm biến CENTER (I2C: 0x30)");

  // --------------------------------------------------------------------------
  // BƯỚC 3: Khởi động cảm biến LEFT (Bên trái)
  // --------------------------------------------------------------------------
  pinMode(XSHUT_LEFT, OUTPUT);
  digitalWrite(XSHUT_LEFT, HIGH);
  delay(450); // Chờ 450ms để tránh sụt áp cộng dồn nguồn 3.3V

  if (!isDevicePresent(0x29)) {
    Serial.println("❌ LỖI VẬT LÝ: Cảm biến LEFT không thức dậy ở địa chỉ 0x29!");
    Serial.println("👉 Vui lòng kiểm tra lại tiếp xúc chân XSHUT_LEFT (GPIO 15) hoặc nguồn VCC/GND!");
    while (1) delay(200);
  }

  if (!lox_left.begin(ADDR_LEFT, false, &Wire)) {
    Serial.println("❌ Lỗi: Không thể cấu hình Left ở địa chỉ 0x31");
    while (1) delay(100);
  }
  Serial.println("✓ Khởi động thành công cảm biến LEFT (I2C: 0x31)");

  // --------------------------------------------------------------------------
  // BƯỚC 4: Khởi động cảm biến RIGHT (Bên phải)
  // --------------------------------------------------------------------------
  pinMode(XSHUT_RIGHT, OUTPUT);
  digitalWrite(XSHUT_RIGHT, HIGH);
  delay(450); // Chờ 450ms cho cảm biến thứ 3 ổn định điện năng hoàn toàn
  if (!lox_right.begin(ADDR_RIGHT, false, &Wire)) {
    Serial.println("❌ Lỗi: Không tìm thấy cảm biến Right ở địa chỉ 0x32");
    while (1) delay(100);
  }
  Serial.println("✓ Khởi động thành công cảm biến RIGHT (I2C: 0x32)");
  
  Serial.println("[SYSTEM] Khởi động thành công toàn bộ 3 cảm biến! 🎉\n");
}

int readDistanceCenter() {
  VL53L0X_RangingMeasurementData_t measure;
  lox_center.rangingTest(&measure, false);
  if (measure.RangeStatus != 4) { 
    return measure.RangeMilliMeter + OFFSET_CENTER;
  }
  return 9999; 
}

int readDistanceLeft() {
  VL53L0X_RangingMeasurementData_t measure;
  lox_left.rangingTest(&measure, false);
  if (measure.RangeStatus != 4) {
    return measure.RangeMilliMeter + OFFSET_LEFT;
  }
  return 9999;
}

int readDistanceRight() {
  VL53L0X_RangingMeasurementData_t measure;
  lox_right.rangingTest(&measure, false);
  if (measure.RangeStatus != 4) {
    return measure.RangeMilliMeter + OFFSET_RIGHT;
  }
  return 9999;
}

void calibrateGyro() {
  Serial.println("[GYRO] Đang hiệu chuẩn con quay hồi chuyển, giữ robot đứng yên...");
  sensors_event_t accel, gyro, temp;
  float sum = 0;
  const int samples = 200; // Giảm số mẫu để hiệu chuẩn nhanh hơn khi mô phỏng

  for (int i = 0; i < samples; i++) {
    mpu.getEvent(&accel, &gyro, &temp);
    sum += gyro.gyro.z;
    delay(2);
  }
  gyroZOffset = sum / samples;
  Serial.print("[GYRO] Hoàn thành. Sai số Offset Z: ");
  Serial.println(gyroZOffset, 6);
}

float readGyroZ() {
  sensors_event_t accel, gyro, temp;
  mpu.getEvent(&accel, &gyro, &temp);
  return gyro.gyro.z - gyroZOffset; 
}

void updateHeading() {
  unsigned long now = millis();
  float dt = (now - lastGyroTime) / 1000.0; 
  lastGyroTime = now;
  
  float gyroZ = readGyroZ(); 
  float gyroZDeg = gyroZ * 57.2957795; 
  
  currentHeading += gyroZDeg * dt;
}

// Giả lập tiến thẳng 1 ô (20cm)
void moveForwardOneBlock() {
  Serial.println("\n>>> [SIMULATION EXECUTE] TIẾN THẲNG 1 Ô (20cm) <<<");
  
  float targetHeading = currentHeading; 
  unsigned long startTime = millis();
  
  // Vẫn chạy vòng lặp để cập nhật log cảm biến và Gyro liên tục trong 800ms
  while (millis() - startTime < MOVE_FORWARD_DURATION) {
    updateHeading();
    
    int distL = readDistanceLeft();
    int distC = readDistanceCenter();
    int distR = readDistanceRight();
    
    // Tính khoảng cách vuông góc thực tế từ thành xe (rộng 12cm) tới tường bên dựa theo góc xiên 45 độ
    // d_perp = Khoảng_Cách_Đo_Được * sin(45°)
    float leftPerp = (distL != 9999) ? (distL * 0.707) : 9999;
    float rightPerp = (distR != 9999) ? (distR * 0.707) : 9999;
    
    // In thông số khoảng cách theo thời gian thực
    static unsigned long lastPrint = 0;
    if (millis() - lastPrint > 150) {
      lastPrint = millis();
      Serial.printf("  [TIẾN THẲNG] Đọc cảm biến: Trái(45°)=%dmm (Vuông góc ~%.1fmm) | Giữa(0°)=%dmm | Phải(45°)=%dmm (Vuông góc ~%.1fmm) | Góc Gyro: %.2f°\n", 
                    distL, leftPerp, distC, distR, rightPerp, currentHeading);
    }
    
    // Giả lập an toàn va chạm
    if (distC < 45) {
      Serial.println("  ⚠️ [SAFETY] Khoảng cách phía trước < 4.5cm. Dừng khẩn cấp!");
      break;
    }
    
    // KHÔNG gọi lệnh setMotor công suất thực tế
    delay(10);
  }
  
  stopMotors();
  delay(500); // Trễ mô phỏng để dễ quan sát trạng thái
}

// Giả lập xoay trái 90 độ
void turnLeft90() {
  Serial.println("\n>>> [SIMULATION EXECUTE] XOAY TRÁI 90 ĐỘ <<<");
  float startHeading = currentHeading;
  float targetHeading = startHeading + 90.0;
  
  unsigned long startTurnTime = millis();
  // Giả lập xoay góc trong 1.5 giây
  while (millis() - startTurnTime < 1200) {
    updateHeading();
    // Giả lập góc xoay tăng dần trên Serial Monitor để dễ quan sát
    float progress = (millis() - startTurnTime) / 1200.0;
    float simulatedHeading = startHeading + (90.0 * progress);
    
    static unsigned long lastPrint = 0;
    if (millis() - lastPrint > 200) {
      lastPrint = millis();
      Serial.printf("  [XOAY TRÁI] Góc xoay hiện tại: %.2f° / %.2f° (Mục tiêu)\n", simulatedHeading, targetHeading);
    }
    delay(10);
  }
  currentHeading = targetHeading; // Đồng bộ hướng giả lập
  Serial.printf("  ✓ Hoàn thành xoay TRÁI. Hướng mới: %.2f°\n", currentHeading);
  delay(500);
}

// Giả lập xoay phải 90 độ
void turnRight90() {
  Serial.println("\n>>> [SIMULATION EXECUTE] XOAY PHẢI 90 ĐỘ <<<");
  float startHeading = currentHeading;
  float targetHeading = startHeading - 90.0;
  
  unsigned long startTurnTime = millis();
  while (millis() - startTurnTime < 1200) {
    updateHeading();
    float progress = (millis() - startTurnTime) / 1200.0;
    float simulatedHeading = startHeading - (90.0 * progress);
    
    static unsigned long lastPrint = 0;
    if (millis() - lastPrint > 200) {
      lastPrint = millis();
      Serial.printf("  [XOAY PHẢI] Góc xoay hiện tại: %.2f° / %.2f° (Mục tiêu)\n", simulatedHeading, targetHeading);
    }
    delay(10);
  }
  currentHeading = targetHeading;
  Serial.printf("  ✓ Hoàn thành xoay PHẢI. Hướng mới: %.2f°\n", currentHeading);
  delay(500);
}

// Giả lập quay đầu 180 độ

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n==================================================================");
  Serial.println("🤖 GIẢ LẬP ROBOCAR GIẢI MÊ CUNG GIÁT THỂ 20x20CM (MOTOR DISABLED)");
  Serial.printf("📏 Thông số xe: Rộng %dmm (12cm) | Dài %dmm (14cm)\n", (int)CAR_WIDTH, (int)CAR_LENGTH);
  Serial.println("📐 Cảm biến Trái/Phải đặt xiên góc 45 độ");
  Serial.println("==================================================================");

  // Khởi tạo Bus I2C
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(30000); // Thiết lập xung nhịp 30kHz cực kỳ ổn định chống nhiễu bus
  delay(200);

  // Khởi tạo MPU6050
  if (!mpu.begin(0x68, &Wire)) {
    Serial.println("❌ Không tìm thấy cảm biến Gyro MPU6050!");
    while (1) delay(100);
  }
  Serial.println("✓ Khởi động thành công MPU6050 (0x68)");

  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  // Khởi tạo 3 cảm biến VL53L0X
  initLaserSensors();
  
  // Cấu hình GPIO Động cơ (Sử dụng cho LED chỉ thị hoặc giữ chân)
  setupMotors();
  
  // Hiệu chuẩn MPU6050
  calibrateGyro();
  
  lastGyroTime = millis();
  currentState = STATE_DECIDE;
  
  Serial.println("🎉 CHẾ ĐỘ GIẢ LẬP ĐÃ SẴN SÀNG! ĐƯA XE VÀO MÊ CUNG ĐỂ CHẠY THỬ LOG...");
  delay(2000);
}

void loop() {
  // Cập nhật tích phân Gyro liên tục
  updateHeading();

  // Đọc khoảng cách thô thực tế từ cảm biến (mm)
  int leftDist   = readDistanceLeft();
  int centerDist = readDistanceCenter();
  int rightDist  = readDistanceRight();

  // Tính toán khoảng cách vuông góc thực tế từ mép ngoài xe (rộng 12cm) tới tường
  // d_perp = d_sensor * sin(45°)
  float leftPerp  = (leftDist != 9999) ? (leftDist * 0.707) : 9999;
  float rightPerp = (rightDist != 9999) ? (rightDist * 0.707) : 9999;

  // Xác định sự tồn tại của tường (sử dụng ngưỡng 45 độ đã phân tích hình học)
  bool wallLeft   = (leftDist < WALL_THRESHOLD_SIDE);
  bool wallCenter = (centerDist < WALL_THRESHOLD_CENTER);
  bool wallRight  = (rightDist < WALL_THRESHOLD_SIDE);

  switch (currentState) {
    
    // TRẠNG THÁI: LỰA CHỌN HƯỚNG ĐI (Thuật toán Bám Tường Trái - Left Hand Rule)
    case STATE_DECIDE: {
      Serial.println("\n========================================================");
      Serial.println("[STATE] TRẠNG THÁI QUYẾT ĐỊNH (STATE_DECIDE)");
      Serial.println("--------------------------------------------------------");
      Serial.printf("🔍 [ĐỌC CẢM BIẾN]  Trái(45°): %dmm | Giữa(0°): %dmm | Phải(45°): %dmm\n", leftDist, centerDist, rightDist);
      Serial.printf("📐 [TÍNH VUÔNG GÓC] Khoảng cách vuông góc hông | Trái: %.1fmm | Phải: %.1fmm\n", leftPerp, rightPerp);
      Serial.printf("🧱 [TƯỜNG VẬT LÝ]  Trái: %s | Giữa: %s | Phải: %s\n", 
                    wallLeft ? "Có Tường (🧱)" : "Trống (🟢)", 
                    wallCenter ? "Có Tường (🧱)" : "Trống (🟢)", 
                    wallRight ? "Có Tường (🧱)" : "Trống (🟢)");

      // Bám Tường Trái:
      if (!wallLeft) {
        Serial.println("👉 [QUYẾT ĐỊNH] => Ưu tiên 1: Bên TRÁI TRỐNG. Xoay TRÁI và đi tới!");
        currentState = STATE_TURN_LEFT;
      } 
      else if (!wallCenter) {
        Serial.println("👉 [QUYẾT ĐỊNH] => Ưu tiên 2: Phía GIỮA TRỐNG. Đi THẲNG!");
        currentState = STATE_MOVE_FORWARD;
      } 
      else if (!wallRight) {
        Serial.println("👉 [QUYẾT ĐỊNH] => Ưu tiên 3: Bên PHẢI TRỐNG. Xoay PHẢI và đi tới!");
        currentState = STATE_TURN_RIGHT;
      } 
      else {
        Serial.println("👉 [QUYẾT ĐỊNH] => Ưu tiên 4: Cụt đường hoàn toàn! QUAY ĐẦU 180°!");
        currentState = STATE_TURN_AROUND;
      }
      break;
    }

    // GIẢ LẬP TIẾN THẲNG 1 Ô
    case STATE_MOVE_FORWARD: {
      moveForwardOneBlock();
      currentState = STATE_DECIDE; 
      break;
    }

    // GIẢ LẬP XOAY TRÁI 90 ĐỘ RỒI TIẾN 1 Ô
    case STATE_TURN_LEFT: {
      turnLeft90();
      moveForwardOneBlock();
      currentState = STATE_DECIDE;
      break;
    }

    // GIẢ LẬP XOAY PHẢI 90 ĐỘ RỒI TIẾN 1 Ô
    case STATE_TURN_RIGHT: {
      turnRight90();
      moveForwardOneBlock();
      currentState = STATE_DECIDE;
      break;
    }

    // GIẢ LẬP QUAY ĐẦU 180 ĐỘ RỒI TIẾN 1 Ô
    case STATE_TURN_AROUND: {
      turnAround180();
      moveForwardOneBlock();
      currentState = STATE_DECIDE;
      break;
    }
  }

  delay(1500); // Dừng lại 1.5 giây giữa các lượt hành động để người dùng dễ theo dõi log
}
