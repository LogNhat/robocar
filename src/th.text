#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Arduino.h>
#include <Wire.h>

Adafruit_MPU6050 mpu;

// ===== PINS =====
const int sensorPins[8] = {36, 39, 34, 35, 32, 33, 25, 26};
const int thresholds[8] = {2953, 3545, 3268, 3368, 3223, 3275, 3452, 3265};
const int weights[8] = {0, 1000, 2000, 3000, 4000, 5000, 6000, 7000};

#define PWMA 27
#define AIN1 18
#define AIN2 19
#define PWMB 13
#define BIN1 5
#define BIN2 23

#define XSHUT_LEFT 15
#define XSHUT_FRONT 16
#define XSHUT_RIGHT 17

// ===== TUNING PD =====
int Kp = 40;
int Kd = 80;
int maxCorrection = 130;

int SPEED_MAX = 135;
int SPEED_FAST = 130;
int SPEED_MID = 120;
int SPEED_CURVE = 120;
int SPEED_SEARCH = 90;

// ===== MPU =====
float gyroZOffset = 0.0;
float gyroZ = 0.0;

// 0 = không in log, 1 = in log MPU + PD
#define DEBUG 1

// ===== STATE =====
int lastError = 0;
int currentSpeed = 100;
int lastDirection = 1;

// ===== MOTOR =====
void motorLeft(int s) {
  s = constrain(s, -255, 255);
  digitalWrite(BIN1, s >= 0);
  digitalWrite(BIN2, s < 0);
  analogWrite(PWMB, abs(s));
}

void motorRight(int s) {
  s = constrain(s, -255, 255);
  digitalWrite(AIN1, s >= 0);
  digitalWrite(AIN2, s < 0);
  analogWrite(PWMA, abs(s));
}

void setMotors(int l, int r) {
  motorLeft(l);
  motorRight(r);
}

// ===== MPU =====
void calibrateGyro() {
  sensors_event_t a, g, temp;
  float sum = 0;

  Serial.println("Calibrating MPU... keep robot still");

  for (int i = 0; i < 500; i++) {
    mpu.getEvent(&a, &g, &temp);
    sum += g.gyro.z;
    delay(2);
  }

  gyroZOffset = sum / 500.0;

  Serial.print("gyroZOffset = ");
  Serial.println(gyroZOffset, 6);
}

float readGyroZ() {
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  return g.gyro.z - gyroZOffset; // rad/s
}

// ===== LINE =====
bool readLine(int &error) {
  long sum = 0;
  int count = 0;

  for (int i = 0; i < 8; i++) {
    if (analogRead(sensorPins[i]) < thresholds[i]) {
      sum += weights[i];
      count++;
    }
  }

  if (!count)
    return false;

  int pos = sum / count;

  // sensor lắp ngược
  error = 3500 - pos;

  if (error > 50)
    lastDirection = 1;
  else if (error < -50)
    lastDirection = -1;

  return true;
}

// ===== SPEED =====
int getSpeed(int error) {
  error = abs(error);

  if (error < 400)
    return SPEED_MAX;
  if (error < 1000)
    return SPEED_FAST;
  if (error < 2200)
    return SPEED_MID;
  return SPEED_CURVE;
}

// ===== CONTROL =====
void followLine() {
  int error;

  // Đọc MPU nhưng chưa điều khiển motor
  gyroZ = readGyroZ();

  if (!readLine(error)) {
    if (lastDirection > 0)
      setMotors(-SPEED_SEARCH, SPEED_SEARCH);
    else
      setMotors(SPEED_SEARCH, -SPEED_SEARCH);
    return;
  }

  int dError = error - lastError;

  int correction = (Kp * error + Kd * dError) / 1000;
  correction = constrain(correction, -maxCorrection, maxCorrection);

  currentSpeed = getSpeed(error);

  int left = constrain(currentSpeed - correction, 0, 255);
  int right = constrain(currentSpeed + correction, 0, 255);

  setMotors(left, right);

  lastError = error;

#if DEBUG
  static unsigned long lastLog = 0;
  if (millis() - lastLog >= 100) {
    lastLog = millis();

    Serial.printf(
        "Err:%5d | Corr:%4d | Spd:%3d | L:%3d | R:%3d | GyroZ:%8.4f\n", error,
        correction, currentSpeed, left, right, gyroZ);
  }
#endif
}

// ===== SETUP =====
void setup() {
  Serial.begin(115200);
  delay(500);

  analogSetWidth(12);
  analogSetAttenuation(ADC_11db);

  pinMode(XSHUT_LEFT, OUTPUT);
  pinMode(XSHUT_FRONT, OUTPUT);
  pinMode(XSHUT_RIGHT, OUTPUT);

  digitalWrite(XSHUT_LEFT, LOW);
  digitalWrite(XSHUT_FRONT, LOW);
  digitalWrite(XSHUT_RIGHT, LOW);

  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);
  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);
  pinMode(PWMA, OUTPUT);
  pinMode(PWMB, OUTPUT);

  for (int i = 0; i < 8; i++) {
    pinMode(sensorPins[i], INPUT);
  }

  setMotors(0, 0);

  Wire.begin(21, 22);

  if (!mpu.begin()) {
    Serial.println("MPU6050 NOT FOUND");
    while (1) {
      setMotors(0, 0);
      delay(1000);
    }
  }

  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  Serial.println("MPU6050 OK");

  calibrateGyro();

  Serial.println("PD + MPU READY");
}

// ===== LOOP =====
void loop() { followLine(); }
